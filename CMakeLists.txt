# FETCHCONTENT_SOURCE_DIR_<NAME> supplies an unpacked local source tree offline.
function(frv_source name url hash)
  FetchContent_Declare(${name} URL "${url}" URL_HASH "SHA256=${hash}")
  FetchContent_GetProperties(${name})
  if(NOT ${name}_POPULATED)
    FetchContent_Populate(${name})
  endif()
  set(${name}_SOURCE_DIR "${${name}_SOURCE_DIR}" PARENT_SCOPE)
  set(${name}_BINARY_DIR "${${name}_BINARY_DIR}" PARENT_SCOPE)
endfunction()

    add_library(fireredvad_${suffix} ${kind} $<TARGET_OBJECTS:frv_objects>)
    set_target_properties(fireredvad_${suffix} PROPERTIES OUTPUT_NAME fireredvad)
    target_include_directories(fireredvad_${suffix} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/native/include> $<INSTALL_INTERFACE:include>)
    target_link_libraries(fireredvad_${suffix} PRIVATE ncnn frv_fbank soxr Threads::Threads)
    set_target_properties(fireredvad_${suffix} PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
    if(kind STREQUAL "SHARED")
      target_link_options(fireredvad_shared PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/native/exports.map")
      if(FRV_WHEEL_LIB_DIR)
        add_custom_command(TARGET fireredvad_shared POST_BUILD
          COMMAND ${CMAKE_COMMAND} -E make_directory ${FRV_WHEEL_LIB_DIR}
          COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:fireredvad_shared> ${FRV_WHEEL_LIB_DIR}/libfireredvad.so)

            COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:soxr> ${FRV_WHEEL_LIB_DIR}/libsoxr.so.0)
        endif()
      endif()
    endif()
    install(TARGETS fireredvad_${suffix} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  endif()
endforeach()
