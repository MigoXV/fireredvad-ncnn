cmake_minimum_required(VERSION 3.20)
project(fireredvad_c_consumer LANGUAGES C)
find_package(FireRedVAD CONFIG REQUIRED)
option(FRV_EXAMPLE_STATIC "Link the static engine archive" OFF)
add_executable(detect detect.c)
if(FRV_EXAMPLE_STATIC)
  target_link_libraries(detect PRIVATE FireRedVAD::static)
else()
  target_link_libraries(detect PRIVATE FireRedVAD::shared)
endif()
